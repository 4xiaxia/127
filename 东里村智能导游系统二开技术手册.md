private selectAgentWithLoadBalancing(preferredAgent: string): string {
  // 策略1: 优先选择指定Agent（负载<80%）
  const targetAgent = this.agents.find(agent => 
    agent.id === preferredAgent && agent.load < 80
  );
  if (targetAgent) return preferredAgent;
  
  // 策略2: 选择负载最低的Agent
  const availableAgents = this.agents.filter(agent => 
    agent.status === 'idle' && agent.load < 60
  );
  
  // 策略3: 考虑响应时间的综合选择
  return availableAgents.reduce((best, current) => {
    const bestScore = (100 - best.load) * 0.7 + (1000 / best.responseTime) * 0.3;
    const currentScore = (100 - current.load) * 0.7 + (1000 / current.responseTime) * 0.3;
    return currentScore > bestScore ? current : best;
  }).id;
}

private async calculateSemanticSimilarity(
  text1: string, 
  text2: string
): Promise<number> {
  // 调用语义分析API
  try {
    const embedding1 = await this.getTextEmbedding(text1);
    const embedding2 = await this.getTextEmbedding(text2);
    return this.cosineSimilarity(embedding1, embedding2);
  } catch (error) {
    console.warn('语义相似度计算失败，降级到传统算法');
    return 0;
  }
}

// 混合相似度计算扩展
private calculateSimilarity(
  str1: string, 
  str2: string, 
  keywords1: string[] = [], 
  keywords2: string[] = []
): number {
  const editSimilarity = this.calculateEditSimilarity(str1, str2);
  const keywordSimilarity = this.calculateKeywordSimilarity(keywords1, keywords2);
  
  // 动态权重配置
  const { weights } = this.config.similarity;
  let similarity = editSimilarity * weights.edit + keywordSimilarity * weights.keyword;
  
  // 可选：加入语义相似度
  if (weights.semantic && weights.semantic > 0) {
    const semanticSimilarity = await this.calculateSemanticSimilarity(str1, str2);
    similarity += semanticSimilarity * weights.semantic;
  }
  
  return similarity;
}

interface TranslationConfig {
  sourceLanguage: string;
  targetLanguage: string;
  provider: 'google' | 'baidu' | 'tencent';
  apiKey: string;
}

interface TranslationRequest {
  text: string;
  sourceLanguage?: string;
  targetLanguage: string;
}

interface TranslationResponse {
  translatedText: string;
  sourceLanguage: string;
  targetLanguage: string;
  confidence: number;
  alternatives?: string[];
}

class TranslationService {
  private static instance: TranslationService;
  private config: TranslationConfig;
  
  static getInstance(): TranslationService {
    if (!TranslationService.instance) {
      TranslationService.instance = new TranslationService();
    }
    return TranslationService.instance;
  }
  
  setConfig(config: TranslationConfig): void {
    this.config = config;
  }
  
  async translate(request: TranslationRequest): Promise<TranslationResponse> {
    try {
      switch (this.config.provider) {
        case 'google':
          return this.translateWithGoogle(request);
        case 'baidu':
          return this.translateWithBaidu(request);
        case 'tencent':
          return this.translateWithTencent(request);
        default:
          throw new Error(`不支持的翻译提供商: ${this.config.provider}`);
      }
    } catch (error) {
      console.error('翻译失败:', error);
      throw error;
    }
  }
  
  private async translateWithGoogle(request: TranslationRequest): Promise<TranslationResponse> {
    // Google翻译API实现
    const response = await fetch('https://translation.googleapis.com/language/translate/v2', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.config.apiKey}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        q: request.text,
        source: request.sourceLanguage || 'auto',
        target: request.targetLanguage,
        format: 'text'
      })
    });
    
    const data = await response.json();
    return {
      translatedText: data.data.translations[0].translatedText,
      sourceLanguage: data.data.translations[0].detectedSourceLanguage,
      targetLanguage: request.targetLanguage,
      confidence: 0.95 // Google API不提供置信度，使用默认值
    };
  }
  
  // 其他翻译提供商实现...
}

interface StorageAdapter {
  get<T>(key: string): Promise<T | null>;
  set<T>(key: string, value: T, options?: StorageOptions): Promise<void>;
  delete(key: string): Promise<void>;
  clear(): Promise<void>;
  keys(): Promise<string[]>;
  size(): Promise<number>;
}

interface StorageOptions {
  ttl?: number;           // 生存时间
  compress?: boolean;      // 是否压缩
  encrypt?: boolean;       // 是否加密
  version?: string;        // 版本号
}

// IndexedDB存储适配器
class IndexedDBAdapter implements StorageAdapter {
  private dbName: string = 'VillageGuideDB';
  private version: number = 1;
  private db: IDBDatabase | null = null;
  
  async initialize(): Promise<void> {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, this.version);
      
      request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        this.db = request.result;
        resolve();
      };
      
      request.onupgradeneeded = (event) => {
        const db = (event.target as IDBOpenDBRequest).result;
        if (!db.objectStoreNames.contains('cache')) {
          db.createObjectStore('cache');
        }
      };
    });
  }
  
  async get<T>(key: string): Promise<T | null> {
    if (!this.db) await this.initialize();
    
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(['cache'], 'readonly');
      const store = transaction.objectStore('cache');
      const request = store.get(key);
      
      request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        const result = request.result;
        if (result && result.expiry > Date.now()) {
          resolve(result.value);
        } else {
          resolve(null);
        }
      };
    });
  }
  
  async set<T>(key: string, value: T, options?: StorageOptions): Promise<void> {
    if (!this.db) await this.initialize();
    
    const item = {
      key,
      value: options?.compress ? this.compress(value) : value,
      expiry: options?.ttl ? Date.now() + options.ttl : Date.now() + 24 * 60 * 60 * 1000,
      version: options?.version || '1.0',
      timestamp: Date.now()
    };
    
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(['cache'], 'readwrite');
      const store = transaction.objectStore('cache');
      const request = store.put(item);
      
      request.onerror = () => reject(request.error);
      request.onsuccess = () => resolve();
    });
  }
  
  // 其他方法实现...
  
  private compress(data: any): string {
    // 实现数据压缩
    return JSON.stringify(data);
  }
}

    } catch (error) {

      if (options?.ttl) {
        await this.client.setex(key, options.ttl / 1000, serialized);
      } else {
        await this.client.set(key, serialized);
      }
    } catch (error) {
      console.error('Redis设置失败:', error);
      throw error;
    }
  }

class StructuredLogger {
  private logLevel: string = process.env.LOG_LEVEL || 'INFO';
  
  private shouldLog(level: string): boolean {
    const levels = ['DEBUG', 'INFO', 'WARN', 'ERROR'];
    return levels.indexOf(level) >= levels.indexOf(this.logLevel);
  }
  
  private formatLog(entry: LogEntry): string {
    return JSON.stringify(entry);
  }
  
  debug(module: string, message: string, data?: any): void {
    if (this.shouldLog('DEBUG')) {
      const entry: LogEntry = {
        timestamp: new Date().toISOString(),
        level: 'DEBUG',
        module,
        message,
        data,
        traceId: this.getTraceId()
      };
      
      console.debug(this.formatLog(entry));
    }
  }
  
  info(module: string, message: string, data?: any): void {
    if (this.shouldLog('INFO')) {
      const entry: LogEntry = {
        timestamp: new Date().toISOString(),
        level: 'INFO',
        module,
        message,
        data,
        traceId: this.getTraceId(),
        userId: this.getUserId()
      };
      
      console.info(this.formatLog(entry));
    }
  }

    if (this.shouldLog('WARN')) {
      const entry: LogEntry = {
        timestamp: new Date().toISOString(),
        level: 'WARN',
        module,
        message,
        data,
        traceId: this.getTraceId(),
        userId: this.getUserId()
      };
      
      console.warn(this.formatLog(entry));
    }

            stack: error.stack
          } : undefined
        },
        traceId: this.getTraceId(),
        userId: this.getUserId()
      };
      
      console.error(this.formatLog(entry));
    }
  }
  
  private getTraceId(): string {

    return '';
  }

// scripts/performanceTest.js

const { performance } = require('perf_hooks');
const axios = require('axios');

class PerformanceTester {
  constructor(baseUrl) {
    this.baseUrl = baseUrl;
    this.results = [];
  }

  async testEndpoint(endpoint, payload, iterations = 100) {
    console.log(`开始测试端点: ${endpoint}`);
    
    const times = [];
    
    for (let i = 0; i < iterations; i++) {
      const startTime = performance.now();
      
      try {
        await axios.post(`${this.baseUrl}${endpoint}`, payload);
        const endTime = performance.now();
        times.push(endTime - startTime);
      } catch (error) {
        console.error(`请求失败:`, error.message);
      }
    }
    
    const avgTime = times.reduce((a, b) => a + b, 0) / times.length;
    const minTime = Math.min(...times);
    const maxTime = Math.max(...times);
    const p95 = this.calculatePercentile(times, 95);
    const p99 = this.calculatePercentile(times, 99);
    
    const result = {
      endpoint,
      iterations,
      avgTime: avgTime.toFixed(2),
      minTime: minTime.toFixed(2),
      maxTime: maxTime.toFixed(2),
      p95: p95.toFixed(2),
      p99: p99.toFixed(2)
    };
    
    this.results.push(result);
    console.log(`测试完成:`, result);
    
    return result;
  }

  calculatePercentile(numbers, percentile) {
    const sorted = numbers.sort((a, b) => a - b);
    const index = Math.ceil((percentile / 100) * sorted.length) - 1;
    return sorted[index];
  }

  generateReport() {
    console.log('\n=== 性能测试报告 ===');
    console.table(this.results);
    
    // 生成HTML报告
    const html = `
      <html>
        <head>
          <title>性能测试报告</title>
          <style>
            table { border-collapse: collapse; width: 100%; }
            th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
            th { background-color: #f2f2f2; }
          </style>
        </head>
        <body>
          <h1>性能测试报告</h1>
          <table>
            <thead>
              <tr>
                <th>端点</th>
                <th>迭代次数</th>
                <th>平均时间(ms)</th>
                <th>最小时间(ms)</th>
                <th>最大时间(ms)</th>
                <th>P95(ms)</th>
                <th>P99(ms)</th>
              </tr>
            </thead>
            <tbody>
              ${this.results.map(result => `
                <tr>
                  <td>${result.endpoint}</td>
                  <td>${result.iterations}</td>
                  <td>${result.avgTime}</td>
                  <td>${result.minTime}</td>
                  <td>${result.maxTime}</td>
                  <td>${result.p95}</td>
                  <td>${result.p99}</td>
                </tr>
              `).join('')}
            </tbody>
          </table>
        </body>
      </html>
    `;
    
    fs.writeFileSync('performance-report.html', html);
    console.log('HTML报告已生成: performance-report.html');
  }
}
